Guessing game basic.

import random

def guessgame():
	number = random.randint(1,100)
	attempts  = 0

	while True:
		try:	
			guess = int(input("Guess a number between 1 and 100: "))
			attempts += 1

			if guess < number:
				print("Too low!")
			elif guess > number:
				print("Too high")
			else:
				print("Correct! Good job!!")
				break

		except ValueError:
			print("Invalid input, insert a number.")

if __name__ == "__main__":
    guessgame()
